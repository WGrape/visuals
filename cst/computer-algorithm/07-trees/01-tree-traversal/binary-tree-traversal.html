# 二叉树遍历 - Python 实现
# ========================================

from typing import Optional, List


class TreeNode:
    """二叉树节点"""
    def __init__(self, val: int = 0):
        self.val = val
        self.left: Optional[TreeNode] = None
        self.right: Optional[TreeNode] = None


class BinaryTree:

    # ── 前序遍历：根 → 左 → 右 ──────────────────
    def preorder(self, root: Optional[TreeNode]) -> List[int]:
        if not root:
            return []
        return [root.val] + self.preorder(root.left) + self.preorder(root.right)

    # ── 前序遍历（迭代版）───────────────────────
    def preorder_iter(self, root: Optional[TreeNode]) -> List[int]:
        if not root:
            return []
        result, stack = [], [root]
        while stack:
            node = stack.pop()
            result.append(node.val)
            if node.right:
                stack.append(node.right)
            if node.left:
                stack.append(node.left)
        return result

    # ── 中序遍历：左 → 根 → 右 ──────────────────
    def inorder(self, root: Optional[TreeNode]) -> List[int]:
        if not root:
            return []
        return self.inorder(root.left) + [root.val] + self.inorder(root.right)

    # ── 中序遍历（迭代版）───────────────────────
    def inorder_iter(self, root: Optional[TreeNode]) -> List[int]:
        result, stack, curr = [], [], root
        while curr or stack:
            while curr:
                stack.append(curr)
                curr = curr.left
            curr = stack.pop()
            result.append(curr.val)
            curr = curr.right
        return result

    # ── 后序遍历：左 → 右 → 根 ──────────────────
    def postorder(self, root: Optional[TreeNode]) -> List[int]:
        if not root:
            return []
        return self.postorder(root.left) + self.postorder(root.right) + [root.val]

    # ── 后序遍历（迭代版）───────────────────────
    def postorder_iter(self, root: Optional[TreeNode]) -> List[int]:
        if not root:
            return []
        result, stack = [], [root]
        while stack:
            node = stack.pop()
            result.append(node.val)
            if node.left:
                stack.append(node.left)
            if node.right:
                stack.append(node.right)
        return result[::-1]


# ── 构建测试树并运行 ─────────────────────────
def build_tree() -> TreeNode:
    #        1
    #      /   \
    #     2     3
    #    / \   / \
    #   4   5 6   7
    root = TreeNode(1)
    root.left  = TreeNode(2)
    root.right = TreeNode(3)
    root.left.left   = TreeNode(4)
    root.left.right  = TreeNode(5)
    root.right.left  = TreeNode(6)
    root.right.right = TreeNode(7)
    return root


if __name__ == "__main__":
    tree = BinaryTree()
    root = build_tree()
    print(f"前序遍历: {tree.preorder(root)}")   # [1, 2, 4, 5, 3, 6, 7]
    print(f"中序遍历: {tree.inorder(root)}")    # [4, 2, 5, 1, 6, 3, 7]
    print(f"后序遍历: {tree.postorder(root)}")  # [4, 5, 2, 6, 7, 3, 1]

// 二叉树遍历 - Go 实现
// ========================================

package main

import "fmt"

// TreeNode 二叉树节点
type TreeNode struct {
    Val   int
    Left  *TreeNode
    Right *TreeNode
}

// NewNode 创建新节点
func NewNode(val int) *TreeNode {
    return &TreeNode{Val: val}
}

// ── 前序遍历：根 → 左 → 右 ──────────────────
func Preorder(root *TreeNode) []int {
    if root == nil {
        return []int{}
    }
    result := []int{root.Val}
    result = append(result, Preorder(root.Left)...)
    result = append(result, Preorder(root.Right)...)
    return result
}

// PreorderIter 前序遍历（迭代版）
func PreorderIter(root *TreeNode) []int {
    if root == nil {
        return nil
    }
    result := []int{}
    stack := []*TreeNode{root}
    for len(stack) > 0 {
        node := stack[len(stack)-1]
        stack = stack[:len(stack)-1]
        result = append(result, node.Val)
        if node.Right != nil {
            stack = append(stack, node.Right)
        }
        if node.Left != nil {
            stack = append(stack, node.Left)
        }
    }
    return result
}

// ── 中序遍历：左 → 根 → 右 ──────────────────
func Inorder(root *TreeNode) []int {
    if root == nil {
        return []int{}
    }
    result := Inorder(root.Left)
    result = append(result, root.Val)
    result = append(result, Inorder(root.Right)...)
    return result
}

// InorderIter 中序遍历（迭代版）
func InorderIter(root *TreeNode) []int {
    result := []int{}
    stack := []*TreeNode{}
    curr := root
    for curr != nil || len(stack) > 0 {
        for curr != nil {
            stack = append(stack, curr)
            curr = curr.Left
        }
        curr = stack[len(stack)-1]
        stack = stack[:len(stack)-1]
        result = append(result, curr.Val)
        curr = curr.Right
    }
    return result
}

// ── 后序遍历：左 → 右 → 根 ──────────────────
func Postorder(root *TreeNode) []int {
    if root == nil {
        return []int{}
    }
    result := Postorder(root.Left)
    result = append(result, Postorder(root.Right)...)
    result = append(result, root.Val)
    return result
}

// PostorderIter 后序遍历（迭代版）
func PostorderIter(root *TreeNode) []int {
    if root == nil {
        return nil
    }
    result := []int{}
    stack := []*TreeNode{root}
    for len(stack) > 0 {
        node := stack[len(stack)-1]
        stack = stack[:len(stack)-1]
        result = append([]int{node.Val}, result...)
        if node.Left != nil {
            stack = append(stack, node.Left)
        }
        if node.Right != nil {
            stack = append(stack, node.Right)
        }
    }
    return result
}

// ── main 函数 ────────────────────────────────
func main() {
    //      1
    //    /   \
    //   2     3
    //  / \   / \
    // 4   5 6   7
    root := NewNode(1)
    root.Left         = NewNode(2)
    root.Right        = NewNode(3)
    root.Left.Left    = NewNode(4)
    root.Left.Right   = NewNode(5)
    root.Right.Left   = NewNode(6)
    root.Right.Right  = NewNode(7)

    fmt.Println("前序遍历:", Preorder(root))   // [1 2 4 5 3 6 7]
    fmt.Println("中序遍历:", Inorder(root))    // [4 2 5 1 6 3 7]
    fmt.Println("后序遍历:", Postorder(root))  // [4 5 2 6 7 3 1]
}

// 二叉树遍历 - PHP 实现
// ========================================

class TreeNode
{
    public int        $val;
    public ?TreeNode $left  = null;
    public ?TreeNode $right = null;

    public function __construct(int $val)
    {
        $this->val = $val;
    }
}

class BinaryTree
{
    // ── 前序遍历：根 → 左 → 右 ──────────────────
    public function preorder(?TreeNode $root): array
    {
        if ($root === null) return [];
        return array_merge(
            [$root->val],
            $this->preorder($root->left),
            $this->preorder($root->right)
        );
    }

    // ── 前序遍历（迭代版）───────────────────────
    public function preorderIter(?TreeNode $root): array
    {
        if ($root === null) return [];
        $result = [];
        $stack  = [$root];
        while (!empty($stack)) {
            $node     = array_pop($stack);
            $result[] = $node->val;
            if ($node->right !== null) $stack[] = $node->right;
            if ($node->left  !== null) $stack[] = $node->left;
        }
        return $result;
    }

    // ── 中序遍历：左 → 根 → 右 ──────────────────
    public function inorder(?TreeNode $root): array
    {
        if ($root === null) return [];
        return array_merge(
            $this->inorder($root->left),
            [$root->val],
            $this->inorder($root->right)
        );
    }

    // ── 中序遍历（迭代版）───────────────────────
    public function inorderIter(?TreeNode $root): array
    {
        $result = [];
        $stack  = [];
        $curr   = $root;
        while ($curr !== null || !empty($stack)) {
            while ($curr !== null) {
                $stack[] = $curr;
                $curr    = $curr->left;
            }
            $curr     = array_pop($stack);
            $result[] = $curr->val;
            $curr     = $curr->right;
        }
        return $result;
    }

    // ── 后序遍历：左 → 右 → 根 ──────────────────
    public function postorder(?TreeNode $root): array
    {
        if ($root === null) return [];
        return array_merge(
            $this->postorder($root->left),
            $this->postorder($root->right),
            [$root->val]
        );
    }

    // ── 后序遍历（迭代版）───────────────────────
    public function postorderIter(?TreeNode $root): array
    {
        if ($root === null) return [];
        $result = [];
        $stack  = [$root];
        while (!empty($stack)) {
            $node     = array_pop($stack);
            $result[] = $node->val;
            if ($node->left  !== null) $stack[] = $node->left;
            if ($node->right !== null) $stack[] = $node->right;
        }
        return array_reverse($result);
    }
}

// ── 构建测试树 ───────────────────────────────
$root               = new TreeNode(1);
$root->left         = new TreeNode(2);
$root->right        = new TreeNode(3);
$root->left->left   = new TreeNode(4);
$root->left->right  = new TreeNode(5);
$root->right->left  = new TreeNode(6);
$root->right->right = new TreeNode(7);

$tree = new BinaryTree();

echo "前序遍历: " . implode(", ", $tree->preorder($root)) . "\n";  // 1,2,4,5,3,6,7
echo "中序遍历: " . implode(", ", $tree->inorder($root))  . "\n";  // 4,2,5,1,6,3,7
echo "后序遍历: " . implode(", ", $tree->postorder($root)) . "\n";  // 4,5,2,6,7,3,1

四种遍历的对比

遍历访问顺序典型用途递归空间迭代空间
前序（根-左-右）根 → 左 → 右复制树、序列化（先建根）、前缀表达式O(h)O(h)
中序（左-根-右）左 → 根 → 右二叉搜索树得到有序序列O(h)O(h)
后序（左-右-根）左 → 右 → 根释放树、计算子树（先处理子节点）O(h)O(h)
层序（BFS）按层从左到右求深度、按层输出、找最短O(w)（队列）O(w)

h = 树高（平衡树为 log n，最坏退化为 n）；w = 最大层宽（最坏 n/2）。所有遍历的时间复杂度都是 O(n)——每个节点访问一次。

实现方式额外空间优点缺点
递归O(h)（调用栈）代码最简洁树深时栈溢出（Go 默认栈 1 GB 但递归深仍危险；C/Java 几千层就爆）
迭代 + 显式栈O(h)无栈溢出风险代码稍复杂
Morris 遍历O(1)零额外空间（利用空指针做临时线索）会临时修改树结构，代码复杂
层序 + 队列O(w)天然按层处理空间可能较大

遍历模板代码

# 递归版本（最简洁）
def inorder(root):
    if not root: return []
    return inorder(root.left) + [root.val] + inorder(root.right)
# 注意：上面用了列表拼接，实际是 O(n^2)；生产用生成器或累加列表

# 迭代版本（推荐，无栈溢出风险）
def inorder_iter(root):
    res, stack, cur = [], [], root
    while cur or stack:
        while cur:                 # 一路向左到底
            stack.append(cur); cur = cur.left
        cur = stack.pop()          # 左子树处理完，访问根
        res.append(cur.val)
        cur = cur.right            # 转向右子树
    return res

# 前序迭代（栈：先压右再压左，弹出顺序才是 根-左-右）
def preorder_iter(root):
    if not root: return []
    res, stack = [], [root]
    while stack:
        node = stack.pop()
        res.append(node.val)
        if node.right: stack.append(node.right)   # 后压右，先弹左
        if node.left:  stack.append(node.left)
    return res

# 后序迭代（技巧：按 根-右-左 收集后反转）
def postorder_iter(root):
    res, stack = [], [root]
    while stack:
        node = stack.pop()
        res.append(node.val)
        if node.left:  stack.append(node.left)
        if node.right: stack.append(node.right)
    return res[::-1]               # 反转得到 左-右-根

常见考点与坑

  • 从遍历序列还原树：① 前序 + 中序可以唯一确定一棵树；② 后序 + 中序也可以；③ 前序 + 后序不能唯一确定（除非是满二叉树）。原理：前/后序确定根，中序划分左右子树。
  • 栈溢出的真实风险：递归深度 = 树高。平衡树 100 万个节点只有 20 层，完全没问题；但退化成链表的树（如按顺序插入 BST）会有 100 万层 → 必爆栈。这就是为什么要平衡树（AVL/红黑树）。
  • Morris 遍历的原理：利用叶子节点的空指针临时指向中序前驱，从而在不使用栈的情况下回到上层。遍历完再把指针还原。空间 O(1) 但每个边会被访问两次（时间仍是 O(n)）。
  • 递归转迭代的通用思路：显式栈模拟调用栈。关键是要想清楚“什么时候访问节点”——前序是入栈时访问，中序是出栈时访问，后序要第二次回到该节点时访问（可用 prev 指针或双栈法）。
  • 应用映射：求树的高度 → 后序（先知道子树高度）；判断两棵树是否相同 → 前序/任意序同步遍历；求从根到叶的路径和 → 前序 + 回溯；找最近公共祖先 LCA → 后序（子树信息向上汇总）。
  • 排障：递归版本栈溢出 → 改迭代；迭代版本死循环 → 检查栈的 pop 与右子树处理顺序；结果顺序不对 → 对照访问顺序定义重新推一遍。
为什么中序遍历二叉搜索树能得到有序序列？

BST 的性质是「左子树所有值 < 根 < 右子树所有值」，而中序的访问顺序是「左 → 根 → 右」，两者正好吻合。所以中序遍历 BST 的输出必然是升序——这是 BST 最核心的性质，也常用于验证一棵树是不是 BST（看中序是否严格递增）。

递归和迭代该写哪个？

面试/教学写递归（清晰）；生产环境写迭代（可控、无栈溢出风险）。如果明确知道树是平衡的（深度 < 100），递归也没问题。Morris 遍历只在明确要求 O(1) 空间时才用。

遍历的工程细节与易错点

递归写法看着简单，但工程上有几个必须知道的细节：递归深度、返回值累积的开销、以及为什么某些题必须用特定遍历顺序。

问题朴素写法更好的写法原因
结果累积开销return pre(left) + [root] + pre(right)传一个 res 切片进去累积+ 每次都新建切片，整体 O(n²)
深度过大递归迭代（显式栈）或 MorrisPython 默认递归上限 1000
需要提前终止递归（要特殊处理返回值）迭代 + 标志位递归里提前 return 容易漏
需要层信息加一个 depth 参数层序 BFS层序天然按层

工程写法

# 高效写法：传入结果切片，避免反复拼接
def inorder(root, res):
    if not root: return
    inorder(root.left, res)
    res.append(root.val)
    inorder(root.right, res)

res = []
inorder(root, res)     # O(n)，无额外分配

# 对比：低效写法
def inorder_bad(root):
    if not root: return []
    return inorder_bad(root.left) + [root.val] + inorder_bad(root.right)
#   n=10000 时，慢 10~100 倍（每层都在拷贝列表）

# 带深度的遍历（求每层最大值、按层处理）
def dfs(node, depth, res):
    if not node: return
    if depth == len(res): res.append([])
    res[depth].append(node.val)
    dfs(node.left, depth + 1, res)
    dfs(node.right, depth + 1, res)

# 提前终止的递归（如找到目标就停）
def find(root, target):
    if not root: return None
    if root.val == target: return root      # 找到就返回，不再往下
    return find(root.left, target) or find(root.right, target)
  • 验证 BST：中序遍历必须是严格递增（有相等就不是合法 BST）。更高效的做法是递归时传上下界 (min, max)，一次遍历搞定，空间 O(h)。
  • 求树高必须是后序（左右子树都算完才能知道当前高度）；求深度从根往下可以前序。
  • 路径类问题用前序 + 回溯：进入节点时 path.append，离开时 path.pop()。忘了 pop 是最常见错误。
  • LCA 用后序：如果左右子树各返回一个非空结果，说明当前节点就是 LCA；如果只有一边非空，就把那个结果往上传递。
  • 序列化：空节点要写占位符（如 #），否则无法还原结构。只用前序遍历（带空占位）就能完整序列化。
为什么很多题必须用后序？

因为后序能自底向上传递信息。求高度、求直径、判断平衡、LCA 这些问题的共同特点是「当前节点的答案依赖左右子树的答案」——必须先知道子树的结果。前序适合「自顶向下传递状态」（如路径和、层数），中序则主要用于 BST。判断用哪个遍历，看信息是从上往下还是从下往上流动。

迭代写法比递归快吗？

通常快一点（省掉函数调用开销和栈帧创建），差距不大（10%~30%），但代码明显更繁琐。真正的优势是不受递归深度限制。建议：默认写递归（可读性好），只有树可能很深（退化成链、n>1000）时才改迭代。Morris 遍历虽是 O(1) 空间，但会临时修改树且常数更大，工程上很少用。

二叉树的遍历能并行吗？

可以——左右子树天然独立，可分到两个线程/协程。但实践中几乎没人这么做：① 树通常很小（几千节点），并行的调度开销远超收益；② 结果合并有额外同步成本。只有在树极大（百万节点）且处理每个节点成本很高（每节点一次 IO 或复杂计算）时才值得；这种场景通常已经改用并行计算框架而不是手写递归。
