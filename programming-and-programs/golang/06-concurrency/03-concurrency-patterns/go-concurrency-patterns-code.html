并发模式不是银弹，关键是把『任务分发 → 并发执行 → 结果收集 → 优雅退出』用 channel 串起来。下面每个模式都给可直接复制运行的骨架。

1工作池 Worker Pool

  • 固定 N 个 worker，从 jobs channel 取任务，写结果到 results。
  • 用 sync.WaitGroup 等所有 worker 结束，防止主协程提前退出。
package main

import (
    "fmt"
    "sync"
)

func main() {
    const workers = 4
    jobs := make(chan int, 100)
    results := make(chan int, 100)
    var wg sync.WaitGroup

    for w := 0; w < workers; w++ {
        wg.Add(1)
        go func(id int) {
            defer wg.Done()
            for j := range jobs { // 通道关闭后自动退出
                results <- j * j
            }
        }(w)
    }

    for i := 1; i <= 20; i++ {
        jobs <- i
    }
    close(jobs) // 通知 worker 不再有任务

    wg.Wait()
    close(results)
    for r := range results {
        fmt.Println(r)
    }
}
要点：不要用 time.Sleep 等 worker 跑完——用 WaitGroup 或 done channel 才可靠。

2流水线 Pipeline

func gen(nums ...int) <-chan int {
    out := make(chan int)
    go func() {
        for _, n := range nums { out <- n }
        close(out)
    }()
    return out
}

func sq(in <-chan int) <-chan int {
    out := make(chan int)
    go func() {
        for n := range in { out <- n * n }
        close(out)
    }()
    return out
}

func main() {
    for n := range sq(sq(gen(2, 3, 4))) { // 两级管道
        fmt.Println(n) // 16 81 256
    }
}
要点：每个阶段都是『接收 channel → 返回 channel』的函数，天然可组合。

3扇出 / 扇入 Fan-out / Fan-in

func fanIn(chans ...<-chan int) <-chan int {
    out := make(chan int)
    var wg sync.WaitGroup
    for _, c := range chans {
        wg.Add(1)
        go func(c <-chan int) {
            defer wg.Done()
            for v := range c { out <- v }
        }(c)
    }
    go func() { wg.Wait(); close(out) }()
    return out
}

// 扇出：同一输入复制到多个 worker；扇入：多个结果汇到一条 channel
a := sq(gen(1, 2))
b := sq(gen(3, 4))
for v := range fanIn(a, b) { fmt.Println(v) }

4限流与 context 取消

  • 限流：用带缓冲 channel 当信号量，或 golang.org/x/time/rate。
  • 取消：把 ctx 透传到每个 goroutine，监听 ctx.Done() 干净退出。
ctx, cancel := context.WithTimeout(context.Background(), 2*time.Second)
defer cancel()

sem := make(chan struct{}, 10) // 最多 10 个并发
for i := 0; i < 100; i++ {
    select {
    case <-ctx.Done():
        return ctx.Err()
    case sem <- struct{}{}:
        go func(i int) {
            defer func() { <-sem }()
            doWork(ctx, i)
        }(i)
    }
}
